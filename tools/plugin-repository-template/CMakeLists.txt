cmake_minimum_required(VERSION 3.24)
project(HotaruPlugins LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Hotaru 插件仓库需要 Windows MSVC x64")
endif()
find_package(Qt6 6.10.2 EXACT REQUIRED COMPONENTS Core Gui Widgets Network)
set(HOTARU_NATIVE_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
set(HOTARU_SPLIT_LAN_METADATA ON)
add_subdirectory(plugins/lan-core)
